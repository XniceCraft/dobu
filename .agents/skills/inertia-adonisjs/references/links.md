# Inertia Links and Navigation (React)

## Link Component

```tsx
import { Link } from "@inertiajs/react";

export default function Navigation() {
  return (
    <nav className="flex gap-4">
      <Link href="/">Home</Link>
      <Link href="/users">Users</Link>
      <Link href="/about">About</Link>
    </nav>
  );
}
```

## Methods and Data

```tsx
<Link href="/logout" method="post" as="button">Logout</Link>
<Link href="/users/1" method="delete">Delete</Link>
<Link href="/users" method="post" data={{ name: "John" }}>Create</Link>
```

## Preserve State and Scroll

```tsx
<Link href="/users?page=2" preserveState preserveScroll>Page 2</Link>
```

## Programmatic Navigation

```tsx
import { router } from "@inertiajs/react";

router.visit("/users");
router.post("/users", { name: "John" });
router.delete("/users/1");
```

## Tuyau (Typed Routes)

Use Tuyau to get end-to-end typed routes in the frontend.

### Where it lives in the repo

- Config: `apps/web/config/tuyau.ts`
- API generated: `apps/web/.adonisjs/api.ts`
- Client: `apps/web/app/core/ui/app/tuyau.ts`

### Generation (required)

Regenerate after API route changes:

```bash
node ace tuyau:generate
```

### Usage with Inertia (React)

```tsx
// inertia/app/app.tsx
import { TuyauProvider } from "@tuyau/inertia/react";
import { tuyau } from "./tuyau";

createInertiaApp({
  setup({ el, App, props }) {
    hydrateRoot(
      el,
      <TuyauProvider client={tuyau}>
        <App {...props} />
      </TuyauProvider>,
    );
  },
});
```

```tsx
import { Link, useRouter } from "@tuyau/inertia/react";

<Link route="users.posts.show" params={{ id: 1, postId: 2 }}>
  Go to post
</Link>;

const router = useRouter();
router.visit("users.posts.show", { id: 1, postId: 2 });
```

### Client setup

```ts
import { createTuyau } from "@tuyau/client";
import { api } from "../../../../.adonisjs/api";

export const tuyau = createTuyau({
  api,
  baseUrl:
    import.meta.env.VITE_API_URL ||
    (typeof window !== "undefined"
      ? window.location.origin
      : "http://localhost:3333"),
  headers: {},
});
```

### Basic calls

```ts
await tuyau.intakes.$post({ /* ... */ });
await tuyau.intakes({ id: "uuid" }).$patch({ /* ... */ });
await tuyau.$route("intakes.messages").$post({ /* ... */ });
const url = tuyau.intakes.$url();
```

### Notes

- `api` is a runtime object generated by `@tuyau/core`.
- For types only, import `ApiDefinition` from `@tuyau/client`.
- Base URL comes from `VITE_API_URL` (fallback: `window.location.origin`).
